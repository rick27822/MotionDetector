cmake_minimum_required(VERSION 3.16)
project(MotionDetectorAnalysis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(analysis_mock INTERFACE)
target_include_directories(analysis_mock INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(mock_demo examples/mock_demo.cpp)
target_link_libraries(mock_demo PRIVATE analysis_mock)
add_executable(mock_tests tests/mock_tests.cpp)
target_link_libraries(mock_tests PRIVATE analysis_mock)
enable_testing()
add_test(NAME analysis_mock_contract COMMAND mock_tests)
add_executable(offline_demo examples/offline_demo.cpp)
target_link_libraries(offline_demo PRIVATE analysis_mock)
add_executable(analysis_tests tests/analysis_tests.cpp)
target_link_libraries(analysis_tests PRIVATE analysis_mock)
add_test(NAME offline_analysis COMMAND analysis_tests)
add_executable(camera_time_tests tests/camera_time_tests.cpp)
target_link_libraries(camera_time_tests PRIVATE analysis_mock)
add_test(NAME camera_time_contract COMMAND camera_time_tests)
