cmake_minimum_required(VERSION 3.21)
project(PoseLabCamera LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network)
find_package(OpenCV 4 REQUIRED COMPONENTS core)
add_library(PoseLabCamera STATIC CameraManager.cpp CameraManager.h ICameraSource.h LatestCameraFrame.h)
set_target_properties(PoseLabCamera PROPERTIES AUTOMOC ON)
target_compile_features(PoseLabCamera PUBLIC cxx_std_17)
target_include_directories(PoseLabCamera PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(PoseLabCamera PUBLIC Qt6::Core Qt6::Gui Qt6::Network opencv_core)
if(WIN32)
    target_link_libraries(PoseLabCamera PRIVATE mf mfplat mfuuid ole32)
endif()
if(PROJECT_IS_TOP_LEVEL OR BUILD_TESTING)
    add_executable(camera_contract_example camera_contract_example.cpp)
    target_link_libraries(camera_contract_example PRIVATE PoseLabCamera)
endif()
